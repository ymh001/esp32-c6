cmake_minimum_required(VERSION 3.16)
project(pocket_ui_test C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../managed_components/lvgl__lvgl lvgl)
add_executable(ui_test ui_test.cpp ../main/ui/shell.cpp ../main/ui/widgets.cpp ../main/apps/apps.cpp ../main/apps/flip_clock.cpp ../main/apps/energy.cpp ../main/fonts/pocket_text_24.c ../main/fonts/flip_tiles.c ../main/fonts/clock_cjk_16.c ../main/fonts/energy_digits_64.c)
target_include_directories(ui_test PRIVATE ../main)
target_link_libraries(ui_test PRIVATE lvgl)
target_compile_options(ui_test PRIVATE -UNDEBUG -Wall -Wextra)

set(POCKET_IDF_PATH "$ENV{IDF_PATH}" CACHE PATH "ESP-IDF checkout (for cJSON parser tests)")
if(EXISTS "${POCKET_IDF_PATH}/components/json/cJSON/cJSON.c")
 add_executable(data_test data_test.cpp ../main/core/energy_parse.cpp "${POCKET_IDF_PATH}/components/json/cJSON/cJSON.c")
 target_include_directories(data_test PRIVATE ../main "${POCKET_IDF_PATH}/components/json/cJSON")
 target_compile_options(data_test PRIVATE -UNDEBUG -Wall -Wextra)
endif()

add_executable(sd_files_test sd_files_test.cpp ../components/board/sd_files.cpp)
target_include_directories(sd_files_test PRIVATE ../components/board/include)
target_compile_options(sd_files_test PRIVATE -UNDEBUG -Wall -Wextra)

add_executable(transfer_files_test transfer_files_test.cpp ../main/core/transfer_files.cpp ../components/board/sd_files.cpp)
target_include_directories(transfer_files_test PRIVATE ../main ../components/board/include)
target_compile_options(transfer_files_test PRIVATE -UNDEBUG -Wall -Wextra)
